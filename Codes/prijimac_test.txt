//Importy pro displej
#include <Wire.h>             // I2C komunikace pro displej
#include <Adafruit_GFX.h>      // Grafická knihovna (kreslení textu, čar)
#include <Adafruit_SSD1306.h>  // Ovládač konkrétního OLED displeje
#include <Fonts/TomThumb.h> //Mikro font

//Importy pro NFC čtečku
#include <SPI.h>               // SPI komunikace pro NFC čtečku
#include <MFRC522.h>           // Knihovna pro čtečku RC522

//Importy pro Wi-Fi a ESP-NOW protokol
#include <WiFi.h>              // Základní Wi-Fi stoh v ESP32
#include <esp_now.h>           // Protokol ESP-NOW

//Deklarace C++ funkcí
#include <unordered_set>

//Tvorba globálních instancí a jejich řídících pinů
#define SCREEN_WIDTH 128
#define SCREEN_HEIGHT 64
Adafruit_SSD1306 display(SCREEN_WIDTH, SCREEN_HEIGHT, &Wire, -1);

#define SS_PIN 5   // Chip Select pro RC522
#define RST_PIN 4  // Reset pin pro RC522
MFRC522 rfc522(SS_PIN, RST_PIN);

//Tvorba tlačítek
#define BUTTON_PIN 16
bool pressed = false;

//Tvorba structu pro přijímaní dat
typedef struct struct_message {
  String name;
  int transmitterID;     // ID vysílače (např. 1 = Anomálie A)
  float radiationPower;    // Základní síla záření vysílače <0.5 - 1>
} struct_message;

struct_message incomingData; // Proměnná pro uložení přijatých dat

int lastRssi = 0;
int peakRssi = -90;
int currentLeader = -1;
bool packetReceived = false;
std::unordered_set<int> bannedAddrs;

//Tyto proměnné se starají o svůj vlastní systém segregace vysílačů
int gotRssi = 0;
int iterations = 0;
int reset = 0;

double radiation = 0;

bool lastRect = true;

// Callback zavolaný automaticky HW přerušením při přijetí zprávy
void OnDataRecv(const esp_now_recv_info *info, const uint8_t *incomingDataPtr, int len) {
  memcpy(&incomingData, incomingDataPtr, sizeof(incomingData));

  if (bannedAddrs.find(incomingData.transmitterID) != bannedAddrs.end()) {
    return;
  }

  lastRssi = info->rx_ctrl->rssi;
  lastRssi *= incomingData.radiationPower;

  if (lastRssi > peakRssi && incomingData.transmitterID != currentLeader) {
    bannedAddrs.insert(incomingData.transmitterID);
    currentLeader = incomingData.transmitterID;
    peakRssi = lastRssi;
  }

  if (incomingData.transmitterID != currentLeader) {
    return;
  }

  packetReceived = true;
  gotRssi = gotRssi + lastRssi;
  iterations++;
}

double calculateAverageRssi() {
  if (iterations == 0) return 0.0;
  return (double)gotRssi / iterations;
}

int calculateRadiationPercent() {
  int avgRssi = calculateAverageRssi();
  if (gotRssi == 0) {
    return 0;
  }

  int clampedRssi = constrain(avgRssi, -90, -40);
  float normalized = (clampedRssi + 90.0f) / 50.0f;
  float curved = pow(normalized, 1.4f);

  return (int)(curved * 100.0f);
}

double calculateDamage(int percent) {
  switch (percent) {
    case 0:          return 0;
    case 1 ... 16:   return 0.005;
    case 17 ... 38:  return 0.1;
    case 39 ... 45:  return 0.6;
    case 46 ... 60:  return 4;
    case 61 ... 80:  return 20;
    case 81 ... 100: return 6;
    default:         return 0;
  }
}

void dealDamage() {
  radiation += calculateDamage(calculateRadiationPercent());
  if (radiation > 100.0) {
    radiation = 100.0;
  }
}

// Převod procent radiace na text úrovně nebezpečí (max 9 znaků, aby nepřetekl řádek)
const char* getDangerLevelText(int percent) {
  if (percent == 0)  return "ZADNE";
  if (percent <= 16) return "ZANEDBATELNE";
  if (percent <= 38) return "NIZKE";
  if (percent <= 45) return "STREDNI";
  if (percent <= 60) return "VYSSI";
  if (percent <= 80) return "VYSOKE";
  return "EXTREMNI";
}

// Vykreslení spodního ukazatele po 10 % (0 dílků = plné zdraví, 10 dílků = smrt)
void drawRadiationBar(double currentRadiation) {
  // Vnější rámeček dole na displeji (x=0, y=40, šířka=128, výška=24)
  display.drawRect(0, 40, 128, 24, WHITE);

  // Výpočet počtu zobrazených obdélníků (po 10 %)
  int activeBlocks = constrain((int)(currentRadiation / 10.0), 0, 10);

  const int startX = 5;   // Levý vnitřní okraj
  const int startY = 44;  // Horní vnitřní okraj
  const int blockW = 10;  // Šířka jednoho dílku
  const int blockH = 16;  // Výška jednoho dílku
  const int stepX  = 12;  // Šířka dílku (10 px) + mezera (2 px)


  int x = startX;
  for (int i = 0; i < activeBlocks; i++) {
    x = startX + (i * stepX);
    display.fillRect(x, startY, blockW, blockH, WHITE);
  }

  if((int) currentRadiation % 10 >= 5 && activeBlocks < 10 && lastRect){
    if(activeBlocks > 0){
      x += stepX;
    }
    display.drawRect(x, startY, blockW, blockH, WHITE);
    lastRect = false;
  }else{
    lastRect = true;
  }
}

// Kompletní překreslení obrazovky PDA
void updateDisplay() {
  display.clearDisplay();
  display.setTextColor(WHITE);

  //Nadpis
  display.setTextSize(1);
  display.setCursor(0, 0);
  display.print("PDA");

  // 2. Prostřední informační řádky
  if (radiation >= 100.0) {
    display.setCursor(0, 16);
    display.println("ZONA TE POHLTILA");
    display.setCursor(0, 25);
    display.println("Vrat se do baru!");
  } else {
    // Řádek o lokaci
    display.setCursor(0, 16);
    display.print("LOKACE: ");
    if (packetReceived) {
      display.println(incomingData.name);
    } else {
      display.println("Hledam...");
    }

    // Řádek o úrovni nebezpečí
    display.setCursor(0, 25);
    display.print("NEBEZPECI: ");
    if (packetReceived) {
      int percent = calculateRadiationPercent();
      display.println(getDangerLevelText(percent));
    } else {
      display.println("---");
    }
  }

  // 3. Spodní bar po 10 %
  drawRadiationBar(radiation);

  display.display();
}

void setup() {
  Serial.begin(115200);
  pinMode(BUTTON_PIN, INPUT_PULLUP);

  if (!display.begin(SSD1306_SWITCHCAPVCC, 0x3C)) {
    Serial.println("OLED displej selhal!");
    for (;;);
  }
  display.clearDisplay();
  display.setTextColor(WHITE);
  display.setTextSize(1);
  display.setCursor(0, 0);
  display.println("Hello, stalker...");
  display.display();

  SPI.begin();
  rfc522.PCD_Init();

  WiFi.mode(WIFI_STA);
  if (esp_now_init() == ESP_OK) {
    esp_now_register_recv_cb(OnDataRecv);
  }
}

void loop() {
  //Kontrola stisknutí tlačítka 
  if(digitalRead(BUTTON_PIN) == LOW && !pressed){
    pressed = true;
    radiation = 0;
  }

  if(digitalRead(BUTTON_PIN) == HIGH && pressed){
    pressed = false;
  }

  Serial.println(radiation);

  // Čtení NFC karty (připraveno pro budoucí léčení / reset radiace)
  if (rfc522.PICC_IsNewCardPresent() && rfc522.PICC_ReadCardSerial()) {
    Serial.print("NFC Tag UID: ");
    for (byte i = 0; i < rfc522.uid.size; i++) {
      Serial.print(rfc522.uid.uidByte[i], HEX);
    }
    Serial.println();
    rfc522.PICC_HaltA();
  }

  // Vykreslení nového UI na OLED displej
  updateDisplay();

  if (reset < 12) {
    reset++;
  } else {
    dealDamage();
    reset = 0;
    gotRssi = 0;
    iterations = 0;
    bannedAddrs.clear();
  }

  delay(250);
}
